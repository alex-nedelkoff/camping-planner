> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the cookie-name identity with real Supabase Auth (magic-link, invite-only), gate the app behind login, key per-person checklists to the authenticated user, and enforce field-level edit permissions (friends edit planning lists/participants/routes; owner edits trip-core + delete).

**Architecture:** FastAPI owns HttpOnly session cookies and mediates Supabase GoTrue over httpx; access-token JWTs are verified locally with pyjwt. An `AUTH_ENABLED` flag (default off) yields a synthetic local user so local dev + the existing suite run without login. `owner_id` lives on the `Trip` model (persisted in the existing JSON blob — no schema change). Authorization is enforced by a `require_user` dependency + per-route owner/field checks.

**Tech Stack:** FastAPI, Jinja2, Pydantic v2, httpx (existing), **pyjwt** (new). Supabase never hit in tests (GoTrue + JWT mocked); `AUTH_ENABLED` off keeps the suite green.

**Conventions:** `python3 -m pytest`. Read mutable config as `from app import config` then `config.X`. Cookies: `cp_at` (access), `cp_rt` (refresh), HttpOnly + SameSite=Lax (Secure in prod).

```python
"""Identity backed by Supabase Auth (Phase 2).

current_user() verifies the access-token JWT from the cp_at cookie. With
AUTH_ENABLED off (local/tests) it returns a synthetic local user so the app
runs without login.
"""
from __future__ import annotations

from dataclasses import dataclass

import jwt
from fastapi import Request, Response

from app import config

ACCESS_COOKIE = "cp_at"
REFRESH_COOKIE = "cp_rt"
_LOCAL = None  # set lazily


@dataclass
class User:
    id: str
    email: str


def _local_user() -> User:
    return User(id="local", email="local")


def _decode(token: str) -> dict | None:
    if not token or not config.SUPABASE_JWT_SECRET:
        return None
    try:
        return jwt.decode(
            token, config.SUPABASE_JWT_SECRET, algorithms=["HS256"],
            audience="authenticated", options={"verify_aud": True},
        )
    except jwt.PyJWTError:
        return None


def current_user(request: Request) -> User | None:
    if not config.AUTH_ENABLED:
        return _local_user()
    claims = _decode(request.cookies.get(ACCESS_COOKIE, ""))
    if not claims:
        return None
    return User(id=claims.get("sub", ""), email=claims.get("email", ""))


def set_session(response: Response, access_token: str, refresh_token: str) -> None:
    common = dict(httponly=True, samesite="lax", secure=True, path="/")
    response.set_cookie(ACCESS_COOKIE, access_token, max_age=3600, **common)
    response.set_cookie(REFRESH_COOKIE, refresh_token, max_age=60 * 60 * 24 * 30, **common)


def clear_session(response: Response) -> None:
    response.delete_cookie(ACCESS_COOKIE, path="/")
    response.delete_cookie(REFRESH_COOKIE, path="/")
```
(Token refresh-on-expiry is wired in Task 5 via the dependency, which can re-issue cookies; `current_user` here returns None on an expired token and the dependency handles refresh.)

"""Supabase GoTrue auth calls (server-side, over httpx)."""
from __future__ import annotations

import httpx

from app import config


def _headers() -> dict:
    return {"apikey": config.SUPABASE_ANON_KEY or "",
            "Content-Type": "application/json"}


def _base() -> str:
    return (config.SUPABASE_URL or "").rstrip("/") + "/auth/v1"


def send_magic_link(email: str, redirect_to: str) -> None:
    r = httpx.post(f"{_base()}/otp",
                   json={"email": email, "options": {"email_redirect_to": redirect_to}},
                   headers=_headers(), timeout=15)
    r.raise_for_status()


def verify_token_hash(token_hash: str, type_: str) -> dict:
    r = httpx.post(f"{_base()}/verify",
                   json={"token_hash": token_hash, "type": type_},
                   headers=_headers(), timeout=15)
    r.raise_for_status()
    return r.json()


def refresh(refresh_token: str) -> dict:
    r = httpx.post(f"{_base()}/token?grant_type=refresh_token",
                   json={"refresh_token": refresh_token},
                   headers=_headers(), timeout=15)
    r.raise_for_status()
    return r.json()
```

"""Magic-link auth routes (server-side)."""
from __future__ import annotations

from fastapi import APIRouter, Form, Request
from fastapi.responses import HTMLResponse, RedirectResponse

from app.services import auth, identity
from app.templating import templates

router = APIRouter()


@router.get("/login", response_class=HTMLResponse)
def login_page(request: Request):
    return templates.TemplateResponse(request, "login.html", {"nav": {"show_user_pill": False}})


@router.post("/login", response_class=HTMLResponse)
def login_submit(request: Request, email: str = Form(...)):
    redirect_to = str(request.base_url).rstrip("/") + "/auth/callback"
    try:
        auth.send_magic_link(email, redirect_to)
    except Exception:
        pass  # don't reveal whether the email exists
    return templates.TemplateResponse(request, "check_email.html",
                                      {"email": email, "nav": {"show_user_pill": False}})


@router.get("/auth/callback")
def auth_callback(request: Request, token_hash: str = "", type: str = "magiclink"):
    resp = RedirectResponse("/", status_code=303)
    try:
        session = auth.verify_token_hash(token_hash, type)
        identity.set_session(resp, session["access_token"], session["refresh_token"])
    except Exception:
        return RedirectResponse("/login?error=1", status_code=303)
    return resp


@router.post("/logout")
def logout():
    resp = RedirectResponse("/login", status_code=303)
    identity.clear_session(resp)
    return resp
```
